=== SUPPRESSION DES COMMENTAIRES VIDES (version corrigée v2) ===

CORRECTION : la v1 ramassait aussi la collection 'notifications' (qui a aussi une
sous-collection 'items'), d'où les erreurs de permissions. Cette v2 ne cible QUE
les documents dont le chemin commence par 'comments/'.

SÉCURITÉ : supprime uniquement les commentaires qui sont TOUS :
  - dans comments/ (jamais notifications/)
  - attachés à un bookId INCONNU
  - PAS sur 'about' ni 'demo-oneshot'
  - VIDES (aucun texte)

------------------------------------------------------------
ÉTAPE A — APERÇU (colle, Entrée) :
------------------------------------------------------------

(async () => {
  const m = await import("https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js");
  const db = m.getFirestore();
  const snap = await m.getDocs(m.collectionGroup(db, "items"));
  const knownPrefixes = ["os-","ine-","sa-","deluca-","apollo-","marine-","reunion-","ecole-","love-"];
  const protectedKeys = ["about","demo-oneshot"];
  const toDelete = [];
  snap.forEach(function(d){
    const parts = d.ref.path.split("/");
    if(parts[0] !== "comments") return;   // <-- IGNORE notifications et tout le reste
    const key = parts[1] || "";
    const data = d.data();
    const km = key.match(/^(.*)_(fr|en)_([0-9]+)$/);
    const bookId = km ? km[1] : key;
    const known = knownPrefixes.some(function(p){ return bookId.indexOf(p) === 0; });
    const isProtected = protectedKeys.indexOf(bookId) !== -1;
    const hasText = !!(data.text && String(data.text).trim());
    if(!known && !isProtected && !hasText){ toDelete.push(d.ref); }
  });
  window.__toDelete = toDelete;
  console.log("A SUPPRIMER (comments uniquement) : " + toDelete.length);
  toDelete.forEach(function(r){ console.log("  - " + r.path); });
  alert("APERCU v2 : " + toDelete.length + " commentaires vides (dans comments/ uniquement).\nSi le nombre te convient, lance l'ETAPE B.");
})();

------------------------------------------------------------
ÉTAPE B — SUPPRESSION (colle après avoir vérifié l'aperçu) :
------------------------------------------------------------

(async () => {
  const m = await import("https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js");
  const db = m.getFirestore();
  if(!window.__toDelete || !window.__toDelete.length){ alert("Lance d'abord l'ETAPE A."); return; }
  let done = 0, fail = 0;
  for(const ref of window.__toDelete){
    try{ await m.deleteDoc(ref); done++; }catch(e){ fail++; console.warn("echec:", ref.path, e.message); }
  }
  console.log("SUPPRIMES: " + done + " | echecs: " + fail);
  alert("Termine !\nSupprimes : " + done + "\nEchecs : " + fail + "\n\nRecharge la page admin.");
})();
