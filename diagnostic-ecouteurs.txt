CODE À COLLER

(() => {
  // Ce test compte les connexions réseau "Listen" (temps réel) ouvertes vers Firestore.
  // Sur un code bien optimisé (lectures ponctuelles), ce nombre reste très bas.
  const entries = performance.getEntriesByType("resource")
    .filter(e => /firestore.googleapis.com/.test(e.name));
  const listen = entries.filter(e => /Listen\/channel/.test(e.name));
  const write  = entries.filter(e => /Write\/channel/.test(e.name));
  const msg =
    "Requêtes Firestore observées sur cette page :\n" +
    "- total: " + entries.length + "\n" +
    "- canaux temps réel (Listen): " + listen.length + "\n" +
    "- canaux d'écriture (Write): " + write.length + "\n\n" +
    (listen.length <= 2
      ? "OK : très peu de temps réel, ton optimisation tient. ✅"
      : "A VERIFIER : plusieurs canaux temps réel ouverts. Envoie ce résultat à l'assistant. ⚠️");
  console.log(msg);
  alert(msg);
})();

------------------------------------------------------------
NOTE
------------------------------------------------------------
- 0 à 2 canaux "Listen" = normal (Firebase en ouvre parfois 1 pour l'auth). Tout va bien.
- Beaucoup de canaux "Listen" qui augmentent quand tu ouvres des chapitres
  = il reste un onSnapshot quelque part -> préviens l'assistant, on le trouvera.
